3. Complete Database Schema (`ticket_booking`)

-- 1. Database Creation
CREATE DATABASE IF NOT EXISTS ticket_booking;
USE ticket_booking;

-- 2. Movies Catalog Table
CREATE TABLE IF NOT EXISTS movies (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(255) NOT NULL,
    genre VARCHAR(100),
    duration INT,
    price DOUBLE
);

-- 3. Confirmed Passenger Bookings Table
CREATE TABLE IF NOT EXISTS bookings (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    movie_id BIGINT,
    customer_name VARCHAR(255),
    seat_number VARCHAR(10),
    theater VARCHAR(255),
    show_date VARCHAR(50),
    show_time VARCHAR(50),
    total_amount DOUBLE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (movie_id) REFERENCES movies(id)
);

4. Seating Matrix Legend

The cinema hall renders a 24-seat layout (Rows A–D, Columns 1–6) with deterministic state feedback:

  • 🟢 AVAILABLE: Seat is open for reservation. Clicking toggles selection.
  • 🟡 SELECTED: Currently picked seat by customer awaiting confirmation.
  • 🔴 OCCUPIED: Locked / reserved seat; disabled from click interaction.

5. Docker Multi-Stage Container Setup

5.1 Backend Multi-Stage Dockerfile (backend/Dockerfile)

# Stage 1: Build the Executable JAR with Maven
FROM maven:3.9-eclipse-temurin-17 AS builder
WORKDIR /app
COPY pom.xml .
RUN mvn dependency:go-offline -B
COPY src ./src
RUN mvn clean package -DskipTests

# Stage 2: Minimalist Lightweight Production Runtime
FROM eclipse-temurin:17-jre-alpine
WORKDIR /app
COPY --from=builder /app/target/*.jar app.jar
EXPOSE 8080
ENTRYPOINT ["java", "-jar", "app.jar"]

5.2 Nginx Reverse Proxy Configuration (frontend/nginx.conf)

server {
    listen 80;
    server_name localhost;

    # Serve static Angular SPA bundles & movie posters
    location / {
        root /usr/share/nginx/html;
        index index.html index.htm;
        try_files $uri$uri/ /index.html;
    }

    # Direct proxying of REST API requests to Spring Boot container
    location /api/ {
        proxy_pass http://backend:8080/api/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}

5.3 Frontend Multi-Stage Dockerfile (frontend/Dockerfile)

# Stage 1: Build Angular Static Artifacts
FROM node:18-alpine AS builder
WORKDIR /app
COPY package*.json ./
RUN npm install
COPY . .
RUN npm run build -- --configuration production

# Stage 2: Serve using Nginx Alpine
FROM nginx:alpine
COPY --from=builder /app/dist/frontend/browser /usr/share/nginx/html
COPY nginx.conf /etc/nginx/conf.d/default.conf
EXPOSE 80
CMD ["nginx", "-g", "daemon off;"]

5.4 Docker Compose Orchestrator (docker-compose.yml)

version: '3.8'

services:
  # 1. Database Tier
  mysql-db:
    image: mysql:8.0
    container_name: cinema-mysql
    restart: always
    environment:
      MYSQL_ROOT_PASSWORD: rootpassword
      MYSQL_DATABASE: ticket_booking
    ports:
      - "3306:3306"
    volumes:
      - db_data:/var/lib/mysql
      - ./backend/src/main/resources/data.sql:/docker-entrypoint-initdb.d/init.sql
    networks:
      - cinema-net

  # 2. Application Server Tier
  backend:
    build:
      context: ./backend
      dockerfile: Dockerfile
    container_name: cinema-backend
    restart: on-failure
    environment:
      SPRING_DATASOURCE_URL: jdbc:mysql://mysql-db:3306/ticket_booking?createDatabaseIfNotExist=true&useSSL=false&allowPublicKeyRetrieval=true
      SPRING_DATASOURCE_USERNAME: root
      SPRING_DATASOURCE_PASSWORD: rootpassword
      SPRING_JPA_HIBERNATE_DDL_AUTO: update
    ports:
      - "8080:8080"
    depends_on:
      - mysql-db
    networks:
      - cinema-net

  # 3. Web & Proxy Tier
  frontend:
    build:
      context: ./frontend
      dockerfile: Dockerfile
    container_name: cinema-frontend
    restart: always
    ports:
      - "80:80"
    depends_on:
      - backend
    networks:
      - cinema-net

volumes:
  db_data:

networks:
  cinema-net:
    driver: bridge

6. Step-by-Step Operations & Terminal Runbook

Deployment Stage Executable PowerShell / Bash Command
1. Build & Start Cluster docker compose up -d --build
2. Inspect Running State docker compose ps
3. Stream Combined Logs docker compose logs -f
4. Stream Backend Logs docker compose logs -f backend
5. Stop Cluster (Preserve Data) docker compose stop
6. Re-Launch Stopped Containers docker compose start
7. Full Teardown docker compose down
8. Teardown + Reset Database docker compose down -v
